{成交量MACD改进指标 - 基于VOL}
SHORT:=12;
LONG:=26;
MID:=9;

{核心计算：将价格替换为成交量}
DIF:EMA(VOL,SHORT)-EMA(VOL,LONG);
DEA:EMA(DIF,MID);
MACD:(DIF-DEA)*2,COLORSTICK;

{以下为颜色逻辑 - 完全参照您提供的价格MACD模式}

{白色线上涨 AND 柱子上涨 -> 红色}
STICKLINE(DIF>REF(DIF,1) AND MACD>REF(MACD,1),MACD,0,1,0),COLORRED;

{白色线下跌 AND 柱子下跌 -> 绿色}
STICKLINE(DIF<REF(DIF,1) AND MACD<REF(MACD,1),MACD,0,1,0),COLORGREEN;

{白色线上涨 AND 柱子下跌 -> 紫色}
STICKLINE(DIF>REF(DIF,1) AND MACD<REF(MACD,1),MACD,0,1,0),COLORMAGENTA;

{白色线下跌 AND 柱子上涨 -> 蓝色}
STICKLINE(DIF<REF(DIF,1) AND MACD>REF(MACD,1),MACD,0,1,0),COLORBLUE;
